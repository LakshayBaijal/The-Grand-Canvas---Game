# Start the game server (Windows). Double-click it, or from PowerShell:
.\start-server.bat

# (macOS / Linux: ./start-server.sh)

# Build the phone app. Reads the server address and Google client id from
# .env (copy .env.example to .env first). Without .env it builds for
# localhost, which is fine for playing on your own Wi-Fi.
.uild-app.ps1              # -> GrandCanvas.apk  (install directly)
.uild-app.ps1 -Bundle      # -> GrandCanvas.aab  (upload to the Play Store)

~/grandcanvas/deploy/update.sh

~/grandcanvas/deploy/reports.sh
